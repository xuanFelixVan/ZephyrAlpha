## 1. 测试策略

### 1.1 测试级别

| 测试级别 | 测试目标 | 测试方法 | 测试工具 |
|----------|----------|----------|----------|
| **单元测试** | 验证单个函数/方法正确性 | 白盒测试、边界值分析 | pytest、unittest |
| **集成测试** | 验证模块间交互正确性 | 接口测试、数据流测试 | pytest、mock |
| **系统测试** | 验证整个成本计算系统 | 端到端测试、场景测试 | pytest、集成环境 |
| **性能测试** | 验证计算性能和资源使用 | 压力测试、负载测试 | pytest-benchmark、memory-profiler |
| **回归测试** | 验证修改不影响现有功能 | 自动化测试套件 | pytest、GitHub Actions |

from cost_calculator import CostCalculator, CostConfig, OrderInfo, OrderSide, MarketType, SecurityCategory

        expected_slippage = 100000 * 0.0002
        assert result.slippage == pytest.approx(expected_slippage, rel=0.01)

    def test_slippage_with_market_data(self):
        """测试带市场数据的滑点计算"""
        config = CostConfig()
        calculator = CostCalculator(config)

        # 高流动性股票
        order1 = OrderInfo(
            symbol="000001.SZ",
            side=OrderSide.BUY,
            quantity=10000,
            price=10.0,
            market=MarketType.SZ,
            category=SecurityCategory.STOCK,
            market_data={
                "daily_volume": 50000000,  # 5000万，高流动性
                "volatility": 0.02,        # 2%波动率，中等
                "avg_trade_size": 10000
            }
        )

        result1 = calculator.calculate_total_cost(order1)
        # 高流动性：因子0.5，中等波动率：因子1.2
        # 基础滑点率0.02% × 0.5 × 1.2 = 0.012%
        expected_slippage1 = 100000 * 0.0002 * 0.5 * 1.2
        assert result1.slippage == pytest.approx(expected_slippage1, rel=0.01)

        # 低流动性股票
        order2 = OrderInfo(
            symbol="000002.SZ",
            side=OrderSide.BUY,
            quantity=10000,
            price=10.0,
            market=MarketType.SZ,
            category=SecurityCategory.STOCK,
            market_data={
                "daily_volume": 500000,    # 50万，低流动性
                "volatility": 0.04,        # 4%波动率，高
                "avg_trade_size": 5000
            }
        )

        result2 = calculator.calculate_total_cost(order2)
        # 低流动性：因子2.0，高波动率：因子1.5
        # 基础滑点率0.02% × 2.0 × 1.5 = 0.06%
        expected_slippage2 = 100000 * 0.0002 * 2.0 * 1.5
        assert result2.slippage == pytest.approx(expected_slippage2, rel=0.01)

            category=SecurityCategory.STOCK,
            market_data={
                "daily_volume": 10000000,
                "volatility": 0.02,
                "avg_trade_size": 10000
            }
        )

        result1 = calculator.calculate_total_cost(order1)
        # 小单：因子1.0

# test_performance_stress.py
import pytest
import time
import threading
from cost_calculator import CostCalculator, CostConfig, OrderInfo, OrderSide, MarketType, SecurityCategory

class TestPerformanceStress:
    """压力测试"""

    def test_high_frequency_calculation(self):
        """测试高频计算"""
        config = CostConfig()
        calculator = CostCalculator(config)

        order = OrderInfo(
            symbol="000001.SZ",
            side=OrderSide.BUY,
            quantity=10000,
            price=10.0,
            market=MarketType.SZ,
            category=SecurityCategory.STOCK
        )

        # 模拟高频交易：1秒内计算10000次
        calculations_per_second = 10000
        duration = 1.0  # 1秒

        start_time = time.perf_counter()
        count = 0

        while time.perf_counter() - start_time < duration:
            calculator.calculate_total_cost(order)
            count += 1

        end_time = time.perf_counter()
        actual_duration = end_time - start_time
        actual_rate = count / actual_duration

        print(f"实际计算速率: {actual_rate:.0f} 次/秒")
        print(f"目标计算速率: {calculations_per_second} 次/秒")
        print(f"完成计算次数: {count}")

        # 验证能够处理高频计算
        assert actual_rate > calculations_per_second * 0.5  # 至少达到目标的50%

    def test_concurrent_stress(self):
        """测试并发压力"""
        config = CostConfig()
        calculator = CostCalculator(config)

        order = OrderInfo(
            symbol="000001.SZ",
            side=OrderSide.BUY,
            quantity=10000,
            price=10.0,
            market=MarketType.SZ,
            category=SecurityCategory.STOCK
        )

        results = []
        lock = threading.Lock()

        def calculate_thread(thread_id, count):
            for i in range(count):
                result = calculator.calculate_total_cost(order)
                with lock:
                    results.append((thread_id, i, result.total_cost))

        # 创建多个线程并发计算
        thread_count = 10
        calculations_per_thread = 1000

        threads = []
        for i in range(thread_count):
            thread = threading.Thread(
                target=calculate_thread,
                args=(i, calculations_per_thread)
            )
            threads.append(thread)

        start_time = time.perf_counter()

        # 启动所有线程
        for thread in threads:
            thread.start()

        # 等待所有线程完成
        for thread in threads:
            thread.join()

        end_time = time.perf_counter()

        total_time = end_time - start_time
        total_calculations = thread_count * calculations_per_thread
        calculations_per_second = total_calculations / total_time

        print(f"并发计算完成: {total_calculations} 次计算")
        print(f"总时间: {total_time:.2f}秒")
        print(f"计算速率: {calculations_per_second:.0f} 次/秒")

        # 验证并发计算正确性
        assert len(results) == total_calculations

        # 所有计算结果应一致
        first_result = results[0][2]
        for _, _, result in results:
            assert result == pytest.approx(first_result, rel=0.01)
