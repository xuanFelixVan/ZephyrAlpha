```python
def generate_mean_reversion_signals(close_prices, window=20, num_std=2):



    """



    ﻝﮔﮒﮒﺙﮒﮒﺛﻛﺟ۰ﮒ?    """



    # ﻟ؟۰ﻝ؟ﮒﺕﮔﮒﺕ?    middle_band = close_prices.rolling(window).mean()



    std = close_prices.rolling(window).std()



    upper_band = middle_band + num_std * std



    lower_band = middle_band - num_std * std



    



    # ﻝﮔﻛﺟ۰ﮒﺓ



    signals = pd.Series(0, index=close_prices.index)



    signals[close_prices <= lower_band] = 1   # ﻛﺗﺍﮒ۴



    signals[close_prices >= upper_band] = -1  # ﮒﮒﭦ



    



    return signals



```

```python
def calculate_risk_parity_weights(returns, target_risk=0.1):



    """



    ﻟ؟۰ﻝ؟ﻠ۲ﻠ۸ﮒﺗﺏﻛﭨﺓﮔﻠ



    """



    # ﻟ؟۰ﻝ؟ﮒﮔﺗﮒﺓ؟ﻝ۸ﻠ?    cov_matrix = returns.cov()



    



# ﻛﺙﮒﻝ؟ﮔﺅﺙﮔﮒﺍﮒﻠ۲ﻠ۸ﻟﺑ۰ﻝ؟ﮒﺓ؟ﮒﺙ



    def objective(weights):



        portfolio_var = np.dot(weights.T, np.dot(cov_matrix, weights))



        marginal_contrib = np.dot(cov_matrix, weights)



        risk_contrib = weights * marginal_contrib / np.sqrt(portfolio_var)



        



# ﻝ؟ﮔﺅﺙﻠ۲ﻠ۸ﻟﺑ۰ﻝ؟ﻝﺕﻝ?        return np.sum((risk_contrib - risk_contrib.mean()) ** 2)



    



    # ﻝﭦ۵ﮔﮔ۰ﻛﭨﭘ



    constraints = [



        {'type': 'eq', 'fun': lambda w: np.sum(w) - 1},  # ﮔﻠﮒﻛﺕﭦ1



        {'type': 'ineq', 'fun': lambda w: w}  # ﮔﻠﻠﻟﺑ



    ]



    



    # ﻛﺙﮒﮔﺎﻟ۶۲



    result = minimize(objective, x0=np.ones(len(returns.columns)) / len(returns.columns),



                     constraints=constraints)



    



    return result.x
```
