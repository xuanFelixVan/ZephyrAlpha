    Returns:
        循环依赖链列�?    """
    circular_deps = []
    visited = set()
    path = []
    
    def dfs(module: str):
        if module in path:
            cycle_start = path.index(module)
            circular_deps.append(path[cycle_start:] + [module])

    return circular_deps
