class ModelGovernance:

    def validate_fairness(

        self,

        model: Any,

        test_data: pd.DataFrame,

        config: FairnessConfig

            sensitive_attributes=config.sensitive_attributes

            predictions, test_data.y, test_data.sensitive_features
