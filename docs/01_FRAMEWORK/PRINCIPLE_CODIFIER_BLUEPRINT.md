from dataclasses import dataclass
from enum import Enum
from typing import Dict, Any, Optional

class Priority(Enum):
    P0 = "P0"
    P1 = "P1"
    P2 = "P2"

class ActionType(Enum):
    ALLOW = "allow"
    WARN = "warn"
    BLOCK = "block"

@dataclass
class PrincipleRule:
    principle_id: str
    principle_name: str
    priority: Priority
    description: str
    
    def check(self, context: Dict[str, Any]) -> bool:
        raise NotImplementedError
    
    def execute(self, context: Dict[str, Any]) -> Dict[str, Any]:
        raise NotImplementedError

class DispersedInvestmentRule(PrincipleRule):
    def __init__(self):
        super().__init__(
            principle_id="CIP-002",
            principle_name="分散投资",
            priority=Priority.P0,
            description="不要把所有鸡蛋放在一个篮子里"
        )
    
    def check(self, context: Dict[str, Any]) -> bool:
        portfolio = context.get("portfolio", {})
        
        single_position_ratio = portfolio.get("single_position_ratio", 0)
        if single_position_ratio > 0.10:
            return False
        
        single_industry_ratio = portfolio.get("single_industry_ratio", 0)
        if single_industry_ratio > 0.30:
            return False
        
        top10_position_ratio = portfolio.get("top10_position_ratio", 0)
        if top10_position_ratio > 0.50:
            return False
        
        return True
    
    def execute(self, context: Dict[str, Any]) -> Dict[str, Any]:
        check_result = self.check(context)
        
        if not check_result:
            return {
                "action": ActionType.BLOCK,
                "reason": f"违反原则 {self.principle_id}: {self.principle_name}",
                "suggestion": "调整持仓结构，降低集中度"
            }
        
        return {
            "action": ActionType.ALLOW,
            "reason": "符合分散投资原则"
        }
```
