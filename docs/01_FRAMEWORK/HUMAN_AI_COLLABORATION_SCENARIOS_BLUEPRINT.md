```python
from dataclasses import dataclass
from typing import Dict, List, Any, Tuple
from enum import Enum


class MarketState(Enum):
    BULL = "bull"
    BEAR = "bear"
    SIDEWAYS = "sideways"
    EXTREME = "extreme"


class StrategyType(Enum):
    TREND_FOLLOWING = "trend_following"
    MEAN_REVERSION = "mean_reversion"
    ARBITRAGE = "arbitrage"
    HEDGING = "hedging"


class TradingScale(Enum):
    SMALL = "small"
    MEDIUM = "medium"
    LARGE = "large"
    EXTRA_LARGE = "extra_large"


class LiquidityLevel(Enum):
    HIGH = "high"
    MEDIUM = "medium"
    LOW = "low"
    CRITICAL = "critical"


@dataclass
class CollaborationScenario:
    market_state: MarketState
    strategy_type: StrategyType
    trading_scale: TradingScale
    liquidity_level: LiquidityLevel
    ai_autonomy: float
    human_involvement: float


class CollaborationSelector:
    def __init__(self, config: Dict[str, Any]):
        self.config = config
        self.scenario_history: List[CollaborationScenario] = []
        
    def select_collaboration_mode(
        self,
        market_state: MarketState,
        strategy_type: StrategyType,
        trading_scale: TradingScale,
        liquidity_level: LiquidityLevel
    ) -> CollaborationScenario:
        market_weights = {
            MarketState.BULL: 0.9,
            MarketState.SIDEWAYS: 0.7,
            MarketState.BEAR: 0.5,
            MarketState.EXTREME: 0.2
        }
        
        strategy_weights = {
            StrategyType.ARBITRAGE: 0.9,
            StrategyType.TREND_FOLLOWING: 0.8,
            StrategyType.MEAN_REVERSION: 0.7,
            StrategyType.HEDGING: 0.6
        }
        
        scale_weights = {
            TradingScale.SMALL: 1.0,
            TradingScale.MEDIUM: 0.8,
            TradingScale.LARGE: 0.5,
            TradingScale.EXTRA_LARGE: 0.3
        }
        
        liquidity_weights = {
            LiquidityLevel.HIGH: 1.0,
            LiquidityLevel.MEDIUM: 0.8,
            LiquidityLevel.LOW: 0.5,
            LiquidityLevel.CRITICAL: 0.2
        }
        
        base_ai_autonomy = 0.7
        
        ai_autonomy = (
            base_ai_autonomy *
            market_weights[market_state] *
            strategy_weights[strategy_type] *
            scale_weights[trading_scale] *
            liquidity_weights[liquidity_level]
        )
        
        ai_autonomy = max(0.1, min(1.0, ai_autonomy))
        
        human_involvement = 1.0 - ai_autonomy
        
        scenario = CollaborationScenario(
            market_state=market_state,
            strategy_type=strategy_type,
            trading_scale=trading_scale,
            liquidity_level=liquidity_level,
            ai_autonomy=ai_autonomy,
            human_involvement=human_involvement
        )
        
        self.scenario_history.append(scenario)
        
        return scenario
```
