### 1.1 因子IC分析体系

> **IC（Information Coefficient�?* 是衡量因子预测能力的核心指标

#### 1.1.1 IC计算公式

| 公式名称 | 数学表达�?| 参数说明 |
|----------|------------|----------|
| **Pearson IC** | $IC_t = \frac{Cov(F_t, R_{t+1})}{\sigma(F_t) \cdot \sigma(R_{t+1})}$ | 因子值与下期收益的相关系�?|
| **Spearman IC（秩IC�?* | $RankIC_t = \rho(Rank(F_t), Rank(R_{t+1}))$ | 基于排名的相关性，更稳�?|

**参数定义**�?
- $F_t$：因子在时刻t的值（横截面标准化后的因子值）
- $R_{t+1}$：t+1时刻的收益率
- $Cov(\cdot)$：协方差
- $\sigma(\cdot)$：标准差
- $\rho(\cdot)$：Spearman秩相关系�?

|--------|----------|----------|

import numpy as np
import pandas as pd
from scipy.stats import pearsonr, spearmanr

def calculate_ic(factor_df: pd.DataFrame, return_df: pd.DataFrame,
                 method: str = 'pearson') -> pd.Series:
    """
    计算因子IC序列

    Parameters:
    -----------
    factor_df : pd.DataFrame
        因子值，index为日期，columns为股票代�?
    return_df : pd.DataFrame
        收益率，index为日期，columns为股票代�?
    method : str
        'pearson' �?'spearman'

    Returns:
    --------
    pd.Series: IC时间序列
    """
    ic_series = []

    for date in factor_df.index:
        factor = factor_df.loc[date]
        future_date = return_df.index[return_df.index.get_loc(date) + 1] if date != return_df.index[-1] else None

        if future_date is None:
            continue

        returns = return_df.loc[future_date]

        valid_idx = factor.index.intersection(returns.index)
        if len(valid_idx) < 30:
            continue

        f = factor[valid_idx].dropna()
        r = returns[valid_idx].dropna()
        common_idx = f.index.intersection(r.index)

        if len(common_idx) < 30:
            continue

        if method == 'pearson':
            ic, _ = pearsonr(f[common_idx], r[common_idx])
        else:
            ic, _ = spearmanr(f[common_idx], r[common_idx])

        ic_series.append({'date': date, 'ic': ic})

    return pd.DataFrame(ic_series).set_index('date')['ic']

def calculate_ir(ic_series: pd.Series) -> dict:
    """
    计算IC_IR等统计量
    """
    ic_mean = ic_series.mean()
    ic_std = ic_series.std()
    ir = ic_mean / ic_std if ic_std != 0 else 0
    win_rate = (ic_series > 0).mean()
    t_stat = ic_mean / (ic_std / np.sqrt(len(ic_series))) if ic_std != 0 else 0

    return {
        'IC均�?: ic_mean,
        'IC标准�?: ic_std,
        'IC_IR': ir,
        'IC胜率': win_rate,
        't统计�?: t_stat,
        '样本�?: len(ic_series)
    }
```

import numpy as np
import pandas as pd

class BarraStyleFactors:
    """Barra风格因子计算"""

    def calculate_size(self, market_cap: pd.Series) -> pd.Series:
        """规模因子：Ln(流通市�?"""
        return np.log(market_cap)

    def calculate_beta(self, returns: pd.DataFrame, market_returns: pd.Series) -> pd.DataFrame:
        """
        市场beta：Cov(Ri,Rm)/Var(Rm)
        """
        variance_m = market_returns.var()
        betas = {}
        for col in returns.columns:
            covariance = returns[col].cov(market_returns)
            betas[col] = covariance / variance_m
        return pd.Series(betas)

    def calculate_momentum(self, prices: pd.DataFrame, lookback: int = 252) -> pd.DataFrame:
        """
        动量因子：过�?2个月收益�?
        """
        return prices.pct_change(periods=lookback)

    def calculate_residual_volatility(self, returns: pd.DataFrame,
                                    market_returns: pd.Series,
                                    betas: pd.Series) -> pd.DataFrame:
        """
        残差波动�?
        """
        residual_returns = returns.sub(market_returns * betas, axis=1)
        return residual_returns.std()

class IndustryFactorModel:
    """行业因子模型"""

    def __init__(self, industry_classification: dict):
        self.industry_class = industry_classification

    def calculate_industry_returns(self, returns: pd.DataFrame,
                                  market_cap: pd.DataFrame) -> pd.DataFrame:
        """
        计算行业市值加权收�?
        """
        industry_returns = {}

        for industry, stocks in self.industry_class.items():
            valid_stocks = [s for s in stocks if s in returns.columns]
            if not valid_stocks:
                continue

            stock_returns = returns[valid_stocks]
            stock_mcap = market_cap[valid_stocks]

            weights = stock_mcap / stock_mcap.sum()
            industry_returns[industry] = (stock_returns * weights).sum(axis=1)

        return pd.DataFrame(industry_returns)

    def calculate_industry_exposure(self, portfolio_weights: pd.Series,
                                  benchmark_weights: pd.Series) -> pd.Series:
        """
        计算行业超配/低配
        """
        industry_exposure = {}

        for industry, stocks in self.industry_class.items():
            valid_stocks = [s for s in stocks if s in portfolio_weights.index]
            if not valid_stocks:
                continue

            port_weight = portfolio_weights[valid_stocks].sum()
            bench_weight = benchmark_weights[valid_stocks].sum()
            industry_exposure[industry] = port_weight - bench_weight

        return pd.Series(industry_exposure)
```
