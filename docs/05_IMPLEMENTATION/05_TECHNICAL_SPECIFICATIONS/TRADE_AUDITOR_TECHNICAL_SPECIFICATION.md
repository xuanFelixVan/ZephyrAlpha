class RiskMonitor:
    """ﻠ۲ﻠ۸ﻝﮔ۶?""
    
    def __init__(self, config: Dict[str, Any]):
        self.config = config
        self.logger = logging.getLogger(__name__)
    
    def monitor_order_risk(
        self,
        order: Dict[str, Any]
    ) -> float:
        """ﻝﮔ۶ﻟ؟۱ﮒﻠ۲ﻠ۸
        
        ﮒﮔﺍ:
            order: ﻟ؟۱ﮒﻛﺟ۰ﮔﺁ
            
        ﻟﺟﮒ:
            ﻠ۲ﻠ۸ﻟﺁﮒ
        """
        risk_score = 0.0
        
        volume = order['volume']
        max_volume = self.config.get('max_volume', 1000000)
        if volume > max_volume:
            risk_score += 0.3
        
        price = order.get('price', 0)
        if price > 0:
            max_price = self.config.get('max_price', 1000.0)
            if price > max_price:
                risk_score += 0.2
        
        return min(risk_score, 1.0)
    
    def monitor_trade_risk(
        self,
        trade: Dict[str, Any]
    ) -> float:
        """ﻝﮔ۶ﮔﻛﭦ۳ﻠ۲ﻠ۸
        
        ﮒﮔﺍ:
            trade: ﮔﻛﭦ۳ﻛﺟ۰ﮔﺁ
            
        ﻟﺟﮒ:
            ﻠ۲ﻠ۸ﻟﺁﮒ
        """
        risk_score = 0.0
        
        slippage = trade.get('slippage', 0)
        max_slippage = self.config.get('max_slippage', 0.05)
        if slippage > max_slippage:
            risk_score += 0.3
        
        commission = trade.get('commission', 0)
        max_commission = self.config.get('max_commission', 1000.0)
        if commission > max_commission:
            risk_score += 0.2
        
        return min(risk_score, 1.0)
