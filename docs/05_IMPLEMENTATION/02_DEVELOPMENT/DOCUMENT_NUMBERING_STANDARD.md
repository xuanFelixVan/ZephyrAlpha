```python
# ﻝ۳ﭦﻛﺝﮔ۲ﮔ۴ﻟﮔ?(ﮒﺁﻠﮔﮒﺍﻝﺍﮔﮒﺓ۴ﮒﺓ)
def check_document_numbering(file_path: str) -> bool:
    """ﮔ۲ﮔ۴ﮔﮔ۰۲ﻝﺙﮒﺓﮒﻟ۶ﮔ?""
    import re
    
    pattern = r'^T\.(\d{2})\.([A-Z]{2,3})(\d{3})\.([a-z0-9_]+)\.md$'
    match = re.match(pattern, os.path.basename(file_path))
    
    if not match:
        return False
    
    layer = match.group(1)  # ﮒﺎﻝﭦ۶
    module_type = match.group(2)  # ﮔ۷۰ﮒﻝﺎﭨﮒ
    sequence = match.group(3)  # ﮒﭦﮒﮒ?
    name = match.group(4)  # ﮔﻟﺟﺍﮒﻝ۶ﺍ
    
    # ﻠ۹ﻟﺁﮒﺎﻝﭦ۶ﮔﮔﮔ?
    valid_layers = ['01', '02', '03', '04', '05', '06', '07', '08', '09']
    if layer not in valid_layers:
        return False
    
    # ﻠ۹ﻟﺁﮔ۷۰ﮒﻝﺎﭨﮒﮔﮔﮔ?
    valid_modules = ['RF', 'DS', 'RM', 'FE', 'SE', 'TE', 'AI', 'AR', 'UT']
    if module_type not in valid_modules:
        return False
    
    return True
```
