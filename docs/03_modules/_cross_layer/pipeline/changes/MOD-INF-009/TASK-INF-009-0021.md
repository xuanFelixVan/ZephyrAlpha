### B151: 熔断器三态机

```python
class PipelineCircuitBreaker:
    state: CircuitBreakerState = CLOSED
    failure_window_s: int = 60
    failure_threshold: int = 3
    cooldown_s: int = 30
    
    async def call(self, fn, *args, **kwargs):
        if self.state == OPEN:
            if time_since_last_failure() > self.cooldown_s:
                self.state = HALF_OPEN  # 探测性放行一次
            else:
                raise CircuitBreakerOpen()
        
        try:
            result = await fn(*args, **kwargs)
            if self.state == HALF_OPEN:
                self.state = CLOSED  # 探测成功恢复
            self._reset_failures()
            return result
        except:
            self._record_failure()
            if self._recent_failures >= self.failure_threshold:
                self.state = OPEN
            raise
```

### B161: $成本追踪

```python
class PipelineCostTracker:
    _cost_total: float = 0.0
    _cost_records: list[CostRecord] = []
    
    def record_call(self, model, module_id, input_tokens, output_tokens):
        rates = {"deepseek": (1.74, 3.48), "glm": (0.0, 0.0), "claude": (5.0, 25.0)}
        cost = (input_tokens/1000)*rates[model][0] + (output_tokens/1000)*rates[model][1]
        self._cost_records.append(CostRecord(model, module_id, input_tokens, output_tokens, cost, time()))
        self._cost_total += cost
    
    def get_cost_summary(self) -> dict:
        """按模型/模块/时间段汇总成本"""
```

class DeadLetterQueue:
    _dead_letters: list[DeadLetterEntry] = []
    max_retries: int = 3
    
    def _maybe_dead_letter(self, task_id, reason, failure_count, error):
        if failure_count >= self.max_retries:
            self._dead_letters.append(DeadLetterEntry(task_id, reason, failure_count, error, time()))
    
    def replay(self, task_id) -> PipelineResult | None:
        """从死信队列取出并重新dispatch"""
