```python
# _call_model 入口
async def _call_model(self, module_id, prompt, ...):
    # L1: 输入检测
    if self._lsg is not None:
        l1_result = self._lsg.check_input(prompt)
        if not l1_result.safe:
            raise LSGInputBlocked(l1_result.reason)
    
    response = await self._api_call(model, prompt)
    
    # L3: 输出检测
    if self._lsg is not None:
        l3_result = self._lsg.check_output(response)
        if not l3_result.safe:
            logger.warning(f"LSG L3 blocked: {l3_result.reason}")
            return ModuleResult(status=FAILURE, errors=[l3_result.reason])
```

### B132: 模型崩塌检测

```python
def _verify_model_diversity(m3_output: str, m7_output: str) -> dict:
    """Jaccard相似度检测——如果M3和M7输出高度同质化→崩塌预警"""
    similarity = jaccard_similarity(tokenize(m3_output), tokenize(m7_output))
    
    if similarity > 0.85:
        return {"collapse_risk": "high", "similarity": similarity,
                "action": "MINORITY_REPORT_ESCALATE"}
    elif similarity > 0.70:
        return {"collapse_risk": "medium", "similarity": similarity,
                "action": "WARN"}
    return {"collapse_risk": "low", "similarity": similarity}
```

### B144: 结构化日志

```python
class PipelineLogger:
    LEVELS = {"DEBUG": 10, "INFO": 20, "WARN": 30, "ERROR": 40}
    _log_buffer: list[LogEntry] = []  # max 2000 entries
    
    def log(self, level: str, module_id: str, message: str, context: dict=None):
        if len(self._log_buffer) >= 2000:  # B148: bounded buffer
            self._log_buffer.pop(0)
        self._log_buffer.append(LogEntry(level, module_id, message, context, time()))
```
