### 1.6 技术指标Python实现

```python
import pandas as pd
import numpy as np

class TechnicalIndicators:
    """技术指标计算"""

    @staticmethod
    def calculate_ma(close: pd.Series, periods: list) -> pd.DataFrame:
        """移动平均线"""
        result = {}
        for p in periods:
            result[f'MA{p}'] = close.rolling(window=p).mean()
        return pd.DataFrame(result)

    @staticmethod
    def calculate_ema(close: pd.Series, periods: list) -> pd.DataFrame:
        """指数移动平均"""
        result = {}
        for p in periods:
            result[f'EMA{p}'] = close.ewm(span=p, adjust=False).mean()
        return pd.DataFrame(result)

    @staticmethod
    def calculate_macd(close: pd.Series, fast: int = 12, slow: int = 26, signal: int = 9) -> pd.DataFrame:
        """MACD指标"""
        ema_fast = close.ewm(span=fast, adjust=False).mean()
        ema_slow = close.ewm(span=slow, adjust=False).mean()
        dif = ema_fast - ema_slow
        dea = dif.ewm(span=signal, adjust=False).mean()
        macd = 2 * (dif - dea)
        return pd.DataFrame({'DIF': dif, 'DEA': dea, 'MACD': macd})

    @staticmethod
    def calculate_rsi(close: pd.Series, periods: list = [6, 12, 24]) -> pd.DataFrame:
        """RSI指标"""
        result = {}
        for p in periods:
            delta = close.diff()
            gain = (delta.where(delta > 0, 0)).rolling(window=p).mean()
            loss = (-delta.where(delta < 0, 0)).rolling(window=p).mean()
            rs = gain / loss
            result[f'RSI{p}'] = 100 - (100 / (1 + rs))
        return pd.DataFrame(result)

    @staticmethod
    def calculate_kdj(high: pd.Series, low: pd.Series, close: pd.Series,
                      n: int = 9, k: int = 3, d: int = 3) -> pd.DataFrame:
        """KDJ指标"""
        lowest_low = low.rolling(window=n).min()
        highest_high = high.rolling(window=n).max()
        rsv = (close - lowest_low) / (highest_high - lowest_low) * 100
        k_value = rsv.ewm(com=(k-1), adjust=False).mean()
        d_value = k_value.ewm(com=(d-1), adjust=False).mean()
        j_value = 3 * k_value - 2 * d_value
        return pd.DataFrame({'K': k_value, 'D': d_value, 'J': j_value})

    @staticmethod
    def calculate_bollinger_bands(close: pd.Series, n: int = 20, k: float = 2) -> pd.DataFrame:
        """布林带"""
        mid = close.rolling(window=n).mean()
        std = close.rolling(window=n).std()
        upper = mid + k * std
        lower = mid - k * std
        return pd.DataFrame({'Upper': upper, 'Mid': mid, 'Lower': lower})

    @staticmethod
    def calculate_atr(high: pd.Series, low: pd.Series, close: pd.Series, n: int = 14) -> pd.Series:
        """ATR指标"""
        tr = pd.concat([
            high - low,
            (high - close.shift()).abs(),
            (low - close.shift()).abs()
        ], axis=1).max(axis=1)
        atr = tr.rolling(window=n).mean()
        return atr

    @staticmethod
    def calculate_obv(close: pd.Series, volume: pd.Series) -> pd.Series:
        """OBV指标"""
        direction = np.sign(close.diff())
        obv = (direction * volume).cumsum()
        return obv
```

### 2.5 形态识别Python实现

```python
import pandas as pd
import numpy as np

class PatternRecognition:
    """形态识别"""

    @staticmethod
    def is_hammer(candle: pd.Series, body_ratio: float = 0.3, shadow_ratio: float = 2.0) -> bool:
        """
        锤子线识别
        candle: [open, high, low, close]
        """
        open_, high, low, close = candle
        body = abs(close - open_)
        total_range = high - low

        if total_range == 0:
            return False

        lower_shadow = min(open_, close) - low
        upper_shadow = high - max(open_, close)

        return (lower_shadow > body * shadow_ratio and
                body < total_range * body_ratio and
                lower_shadow > upper_shadow)

    @staticmethod
    def is_engulfing( candles: pd.DataFrame) -> str:
        """
        吞没形态识别
        candles: DataFrame with [open, high, low, close]
        返回: 'bullish', 'bearish', or None
        """
        if len(candles) < 2:
            return None

        prev = candles.iloc[-2]
        curr = candles.iloc[-1]

        prev_bearish = prev['close'] < prev['open']
        curr_bullish = curr['close'] > curr['open']

        curr_contains_prev = (curr['low'] <= prev['low'] and curr['high'] >= prev['high'])

        if prev_bearish and curr_bullish and curr_contains_prev:
            return 'bullish'

        prev_bullish = prev['close'] > prev['open']
        curr_bearish = curr['close'] < curr['open']

        if prev_bullish and curr_bearish and curr_contains_prev:
            return 'bearish'

        return None

    @staticmethod
    def detect_bollinger_band_squeeze(close: pd.Series, window: int = 20, num_std: float = 2.0) -> bool:
        """
        布林带收窄识别（潜在突破信号）
        """
        mid = close.rolling(window=window).mean()
        std = close.rolling(window=window).std()
        bandwidth = (mid + num_std * std - (mid - num_std * std)) / mid
        return bandwidth < 0.1
```

### 3.2 复盘Python实现

```python
import pandas as pd
import numpy as np

class TradingReview:
    """交易复盘分析"""

    def __init__(self, trades: pd.DataFrame, prices: pd.DataFrame):
        """
        trades: 交易记录 [date, code, direction, volume, price]
        prices: 价格数据 [date, code, close]
        """
        self.trades = trades
        self.prices = prices

    def calculate_daily_returns(self) -> pd.Series:
        """计算每日收益率"""
        daily_pnl = self.trades.groupby('date')['pnl'].sum()
        return daily_pnl / self.trades.groupby('date')['cost'].sum()

    def calculate_cumulative_returns(self, initial_capital: float) -> pd.Series:
        """计算累计收益率"""
        daily_returns = self.calculate_daily_returns()
        cumulative = (1 + daily_returns).cumprod() - 1
        return cumulative * initial_capital

    def calculate_max_drawdown(self, cumulative_returns: pd.Series) -> float:
        """计算最大回撤"""
        peak = cumulative_returns.expanding(min_periods=1).max()
        drawdown = (cumulative_returns - peak) / peak
        return drawdown.min()

    def calculate_sharpe_ratio(self, daily_returns: pd.Series,
                               risk_free_rate: float = 0.03) -> float:
        """计算夏普比率"""
        excess_returns = daily_returns - risk_free_rate / 252
        return np.sqrt(252) * excess_returns.mean() / excess_returns.std()

    def calculate_win_rate(self) -> float:
        """计算胜率"""
        profitable_trades = self.trades[self.trades['pnl'] > 0]
        return len(profitable_trades) / len(self.trades) if len(self.trades) > 0 else 0

    def calculate_profit_loss_ratio(self) -> float:
        """计算盈亏比"""
        avg_profit = self.trades[self.trades['pnl'] > 0]['pnl'].mean()
        avg_loss = abs(self.trades[self.trades['pnl'] < 0]['pnl'].mean())
        return avg_profit / avg_loss if avg_loss != 0 else 0

    def generate_review_report(self, initial_capital: float = 1000000) -> dict:
        """生成复盘报告"""
        daily_returns = self.calculate_daily_returns()
        cumulative = self.calculate_cumulative_returns(initial_capital)

        return {
            '总交易次数': len(self.trades),
            '胜率': self.calculate_win_rate(),
            '盈亏比': self.calculate_profit_loss_ratio(),
            '累计收益率': cumulative.iloc[-1] if len(cumulative) > 0 else 0,
            '最大回撤': self.calculate_max_drawdown(cumulative),
            '夏普比率': self.calculate_sharpe_ratio(daily_returns),
            '日均收益率': daily_returns.mean(),
            '日收益率标准差': daily_returns.std()
        }
```

```python
class LimitUpAnalyzer:
    """涨停板量化分析"""

    def __init__(self, market_data: pd.DataFrame):
        self.data = market_data

    def calculate_limit_up_score(self, stock: pd.Series) -> float:
        """
        计算涨停板综合评分
        """
        score = 0

        # 连板加分
        if stock.get('连续板数', 0) >= 5:
            score += 40
        elif stock.get('连续板数', 0) >= 3:
            score += 30
        elif stock.get('连续板数', 0) >= 2:
            score += 20

        # 封单比加分
        seal_ratio = stock.get('封单比', 0)
        if seal_ratio > 2:
            score += 20
        elif seal_ratio > 1:
            score += 15
        elif seal_ratio > 0.5:
            score += 10

        # 首封时间加分
        first_seal_time = stock.get('首封时间', '')
        if first_seal_time and first_seal_time < '09:45':
            score += 20
        elif first_seal_time and first_seal_time < '10:00':
            score += 15
        elif first_seal_time and first_seal_time < '10:30':
            score += 10

        # 一字板加分
        if stock.get('开板次数', 1) == 0:
            score += 10

        # 换手率加分
        turnover = stock.get('换手率', 0)
        if turnover > 15:
            score += 10
        elif turnover > 10:
            score += 5

        return score

    def screen_limit_up_stocks(self, date: str,
                               min_连续板数: int = 2,
                               max_流通值: float = 100e8,
                               min_换手率: float = 5,
                               min_封单比: float = 0.5) -> pd.DataFrame:
        """
        筛选连板股
        """
        stocks = self.data[self.data['日期'] == date].copy()

        filtered = stocks[
            (stocks['连续板数'] >= min_连续板数) &
            (stocks['流通值'] <= max_流通值) &
            (stocks['换手率'] > min_换手率) &
            (stocks['封单比'] > min_封单比)
        ]

        filtered['综合评分'] = filtered.apply(
            self.calculate_limit_up_score, axis=1
        )

        return filtered.sort_values('综合评分', ascending=False)

    def calculate_market_heat(self, date: str) -> dict:
        """
        计算市场热度指标
        """
        stocks = self.data[self.data['日期'] == date]

        limit_up_count = len(stocks[stocks['涨跌幅'] >= 9.9])
        limit_down_count = len(stocks[stocks['涨跌幅'] <= -9.9])
        board_stock_count = len(stocks[stocks['连续板数'] >= 2])

        broken_count = len(stocks[
            (stocks['涨跌幅'] >= 9.9) &
            (stocks['开板次数'] > 0)
        ])
        burst_rate = broken_count / limit_up_count if limit_up_count > 0 else 0

        yesterday_limit_up = stocks[
            (stocks['日期'] == self.get_previous_trade_date(date)) &
            (stocks['涨跌幅'] >= 9.9)
        ]
        premium = yesterday_limit_up['今日涨幅'].mean() if len(yesterday_limit_up) > 0 else 0

        return {
            '涨停总数': limit_up_count,
            '跌停总数': limit_down_count,
            '连板数': board_stock_count,
            '炸板率': burst_rate,
            '昨日涨停溢价': premium
        }
```

```python
class NewsSentimentAnalyzer:
    """新闻情绪分析"""

    def __init__(self):
        self.sentiment_model = None
        self.keywords_positive = ['超预期', '突破', '增长', '龙头', '看好']
        self.keywords_negative = ['风险', '下调', '亏损', '利空', '警示']

    def calculate_sentiment_score(self, news_text: str) -> float:
        """计算新闻情感得分"""
        positive_count = sum(1 for kw in self.keywords_positive if kw in news_text)
        negative_count = sum(1 for kw in self.keywords_negative if kw in news_text)

        total = positive_count + negative_count
        if total == 0:
            return 0.0

        return (positive_count - negative_count) / total

    def calculate_news_intensity(self, news_list: list) -> dict:
        """计算新闻情绪强度"""
        positive = sum(1 for n in news_list if self.calculate_sentiment_score(n) > 0)
        negative = sum(1 for n in news_list if self.calculate_sentiment_score(n) < 0)

        intensity = positive / (positive + negative) if (positive + negative) > 0 else 0.5

        return {
            'positive_count': positive,
            'negative_count': negative,
            'intensity': intensity,
            'sentiment': 'positive' if intensity > 0.6 else 'negative' if intensity < 0.4 else 'neutral'
        }

    def analyze_stock_news(self, stock_code: str, news_data: list) -> dict:
        """分析个股新闻情绪"""
        scores = [self.calculate_sentiment_score(n['content']) for n in news_data]
        intensity = self.calculate_news_intensity([n['content'] for n in news_data])

        return {
            'stock_code': stock_code,
            'avg_sentiment': sum(scores) / len(scores) if scores else 0,
            'sentiment_count': len(scores),
            'intensity': intensity,
            'signal': 'BUY' if intensity['intensity'] > 0.6 else 'SELL' if intensity['intensity'] < 0.4 else 'HOLD'
        }
```

```python
class AnalystExpectationAnalyzer:
    """分析师预期分析"""

    def __init__(self):
        self.buy_threshold = 0.6
        self.upside_threshold = 0.30

    def calculate_earning_revision(self, current_expectation: float,
                                   previous_expectation: float) -> float:
        """计算盈利预期调整"""
        if previous_expectation == 0:
            return 0.0
        return (current_expectation - previous_expectation) / previous_expectation

    def analyze_rating_distribution(self, ratings: list) -> dict:
        """分析评级分布"""
        buy = sum(1 for r in ratings if r in ['买入', '强烈推荐', '增持'])
        hold = sum(1 for r in ratings if r in ['持有', '中性', '谨慎'])
        sell = sum(1 for r in ratings if r in ['卖出', '减持', '回避'])

        total = len(ratings)
        if total == 0:
            return {'buy_ratio': 0, 'hold_ratio': 0, 'sell_ratio': 0}

        return {
            'buy_ratio': buy / total,
            'hold_ratio': hold / total,
            'sell_ratio': sell / total,
            'signal': 'BUY' if buy/total > self.buy_threshold else 'HOLD'
        }

    def calculate_target_upside(self, target_price: float, current_price: float) -> float:
        """计算目标价空间"""
        return (target_price - current_price) / current_price

    def analyze_expectation(self, stock_data: dict) -> dict:
        """综合分析师预期分析"""
        earning_rev = self.calculate_earning_revision(
            stock_data.get('current_eps_expectation', 0),
            stock_data.get('previous_eps_expectation', 0)
        )

        rating_dist = self.analyze_rating_distribution(stock_data.get('ratings', []))

        target_upside = self.calculate_target_upside(
            stock_data.get('avg_target_price', 0),
            stock_data.get('current_price', 0)
        )

        return {
            'earning_revision': earning_rev,
            'rating_distribution': rating_dist,
            'target_upside': target_upside,
            'signal': 'BUY' if (earning_rev > 0.1 and target_upside > self.upside_threshold) else 'HOLD'
        }
```

```python
    def get_investment_response(self, inflation_regime: str) -> dict:
        """获取通胀环境下的投资建议"""
        response_map = {
            '通缩': {'债券': 0.60, '股票': 0.20, '商品': 0.10, '现金': 0.10},
            '温和通胀': {'债券': 0.30, '股票': 0.40, '商品': 0.15, '现金': 0.15},
            '较高通胀': {'债券': 0.15, '股票': 0.35, '商品': 0.35, '现金': 0.15},
            '高通胀': {'债券': 0.10, '股票': 0.20, '商品': 0.50, '现金': 0.20}
        }
        return response_map.get(inflation_regime, {})

    def analyze(self, macro_data: dict) -> dict:
        """综合通胀分析"""
        cpi = macro_data.get('cpi', 0)
        ppi = macro_data.get('ppi', 0)
        core_cpi = macro_data.get('core_cpi', 0)

        regime = self.get_inflation_regime(cpi, ppi, core_cpi)
        response = self.get_investment_response(regime)

        return {
            'inflation_regime': regime,
            'cpi': cpi,
            'ppi': ppi,
            'core_cpi': core_cpi,
            'recommended_allocation': response,
            'bond_impact': 'negative' if cpi > self.high_inflation else 'positive'
        }
```
