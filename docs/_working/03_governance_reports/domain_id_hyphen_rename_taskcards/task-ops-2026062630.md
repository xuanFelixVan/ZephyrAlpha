### V1：生成制品中 6 个新域名出现

**验证命令**：
```bash
python -c "
import os
new_ids = ['D-GOV_DOCS','D-GOV_ENFORCEMENT','D-GOV_SCRIPTS','D-AUDITTEST','D-INTEGRATION_GATEWAY','D-SECURITY_LLM']
dirs = [
    'docs/02_enterprise_architecture/02_domain_architecture_docs',
    'docs/02_enterprise_architecture/01_global_architecture_diagram',
    'docs/02_enterprise_architecture/generated',
]
files = [
    'data/asset_index/target_path_tree.yaml',
    'data/asset_index/project_entity_depgraph.yaml',
    'docs/02_enterprise_architecture/03_governance_reports/design_vs_production.md',
    'docs/02_enterprise_architecture/03_governance_reports/capacity_report.md',
]
all_found = {nid: False for nid in new_ids}
def scan(content):
    for nid in new_ids:
        if nid in content:
            all_found[nid] = True
for f in files:
    if os.path.exists(f):
        scan(open(f, encoding='utf-8', errors='ignore').read())
for d in dirs:
    if os.path.isdir(d):
        for root, _, fs in os.walk(d):
            for fn in fs:
                if fn.endswith(('.md','.yaml','.mmd')):
                    try: scan(open(os.path.join(root,fn), encoding='utf-8', errors='ignore').read())
                    except: pass
for nid, found in all_found.items():
    print(f'{\"PASS\" if found else \"FAIL\"}: {nid} found in artifacts')
"
```

**通过判定**：6 行全 PASS（每个新域名至少在 1 个生成制品中出现）。

**FAIL 修复**：若有新域名未出现，说明对应生成器未运行或未覆盖该域。回 OPS-2026062629 对应动作重跑生成器。

### V2：生成制品中 6 个旧域名零残留

**验证命令**：
```bash
python -c "
import os, re
old_ids = ['D-GOV-DOCS','D-GOV-ENFORCEMENT','D-GOV-SCRIPTS','D-GOV_AUDIT_TESTS','D-INTEGRATION-GATEWAY','D-SECURITY-LLM']
pat = re.compile('|'.join(r'(?<![A-Z])' + re.escape(d) for d in old_ids))
dirs = [
    'docs/02_enterprise_architecture/02_domain_architecture_docs',
    'docs/02_enterprise_architecture/01_global_architecture_diagram',
    'docs/02_enterprise_architecture/generated',
]
files = [
    'data/asset_index/target_path_tree.yaml',
    'data/asset_index/project_entity_depgraph.yaml',
    'docs/02_enterprise_architecture/03_governance_reports/design_vs_production.md',
    'docs/02_enterprise_architecture/03_governance_reports/capacity_report.md',
]
total = 0
def scan(path):
    global total
    try:
        content = open(path, encoding='utf-8', errors='ignore').read()
        for m in pat.finditer(content):
            if '[BLUEPRINT]' in content[max(0,m.start()-50):m.start()+50]:
                continue
            print(f'  RESIDUAL: {path}: {m.group()}')
            total += 1
    except: pass
for f in files:
    if os.path.exists(f): scan(f)
for d in dirs:
    if os.path.isdir(d):
        for root, _, fs in os.walk(d):
            for fn in fs:
                if fn.endswith(('.md','.yaml','.mmd')):
                    scan(os.path.join(root,fn))
print(f'{\"PASS\" if total == 0 else \"FAIL\"}: residual={total}')
"
```

**通过判定**：`PASS: residual=0`，无 RESIDUAL 行。

**FAIL 修复**：若有残留，说明生成器仍引用旧域名。检查生成器脚本硬编码是否已改（回 OPS-2026062624 V2/V3/V4 验证），或 DB 是否有残留（回 OPS-2026062622 V4 验证）。
